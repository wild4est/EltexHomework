```bash
#   _____ _____ __ __ _____
#\\\\  __\\  __\\ \\ \\    \
#\\\\\  __\\ \___\  _ \\  \ \
#\\\\\\____\\____\\_\\_\\____\ server


# == Общее ==
# Сборка сервера и клиента
make all

# Очистка сервера и клиента
make clean_all

# Во время запуска сервера ему необходио задать порт и сетевой адрес
# ./server_main <порт сервера> <сетевой дрес сервера>
sudo ./server_main 1234 0.0.0.0

# Во время запуска клиента ему необходимо задать порт и сетевой адрес, а так же указать порт и адрес сервера
# ./client_main <порт клиента> <порт сервера> <сетевой адрес клиента> <сетевой адрес сервера>
sudo ./client_main 1234 1234 0.0.0.0 0.0.0.0

# == Сервер ==
# Сборка сервера
make compile_server

# Очистка сервера
make clean_server

# === Клиент ===
# Сборка клиента
make compile_client

# Очистка клиента
make clean_client

```
